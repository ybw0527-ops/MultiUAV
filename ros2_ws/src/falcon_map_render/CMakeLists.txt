cmake_minimum_required(VERSION 3.8)
project(falcon_map_render LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(assimp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(render_core SHARED src/render_core.cpp)
target_include_directories(render_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(render_core SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(render_core PUBLIC Eigen3::Eigen PRIVATE ${PCL_LIBRARIES} assimp::assimp)
target_compile_options(render_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(map_render_node src/map_render_node.cpp)
add_executable(check_clearance src/check_clearance.cpp)
target_link_libraries(check_clearance render_core)
target_link_libraries(map_render_node render_core)
ament_target_dependencies(map_render_node rclcpp nav_msgs sensor_msgs geometry_msgs ament_index_cpp)
install(TARGETS render_core map_render_node check_clearance EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
# Reuse upstream assets, with no duplicate 100 MB map copy in source control.
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../uav_simulator/map_render/resource/" DESTINATION share/${PROJECT_NAME}/resource)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_render_core test/test_render_core.cpp)
  target_link_libraries(test_render_core render_core)
  target_compile_definitions(test_render_core PRIVATE FALCON_MAP_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../../uav_simulator/map_render/resource")
endif()
ament_package()
