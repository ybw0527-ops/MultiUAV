cmake_minimum_required(VERSION 3.8)
project(falcon_exploration_manager)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp falcon_exploration_preprocessing falcon_fast_planner falcon_interfaces falcon_trajectory falcon_exploration_utils nav_msgs geometry_msgs std_msgs visualization_msgs)
foreach(dependency ${dependencies})
  find_package(${dependency} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/exploration_manager.cpp src/exploration_fsm.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${dependencies})
add_executable(exploration_node src/exploration_node.cpp)
target_link_libraries(exploration_node ${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} exploration_node EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_exploration_manager test/test_exploration_manager.cpp)
  target_link_libraries(test_exploration_manager ${PROJECT_NAME})
endif()
ament_package()
