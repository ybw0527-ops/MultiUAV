cmake_minimum_required(VERSION 3.8)
project(falcon_so3_simulator)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED)
find_package(rclcpp REQUIRED)
find_package(falcon_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(quadrotor_dynamics src/Quadrotor.cpp)
target_include_directories(quadrotor_dynamics PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${Boost_INCLUDE_DIRS})
target_link_libraries(quadrotor_dynamics PUBLIC Eigen3::Eigen)
add_executable(so3_simulator_node src/so3_simulator_node.cpp)
target_link_libraries(so3_simulator_node quadrotor_dynamics)
ament_target_dependencies(so3_simulator_node rclcpp falcon_interfaces nav_msgs sensor_msgs std_msgs geometry_msgs)
install(TARGETS quadrotor_dynamics EXPORT export_dynamics ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS so3_simulator_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(FILES test/tracking_run_summary.json test/execution_run_summary.json DESTINATION share/${PROJECT_NAME}/test)
ament_export_targets(export_dynamics HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 Boost)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dynamics test/test_dynamics.cpp)
  target_link_libraries(test_dynamics quadrotor_dynamics)
endif()
ament_package()
